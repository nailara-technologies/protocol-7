# name  = model_batch.diff.text
# descr = lazy line diff between two blob checksums, for review drill-down
# note  = compute-on-read : no diff is ever stored at capture time.
#         line sets via base.diff_array [ added/removed, not positional ]
# param = <before-checksum> <after-checksum> -> <diff-string> | undef

my $before_cs = shift // '';
my $after_cs  = shift // '';

if ( !length $before_cs or !length $after_cs ) {
    warn 'expected two checksum parameters <{C1}>';
    return undef;
}

my $before_content = <[model_batch.blob.read]>->($before_cs);
my $after_content  = <[model_batch.blob.read]>->($after_cs);

if ( !defined $before_content or !defined $after_content ) {
    warn 'blob content unavailable for diff <{C1}>';
    return undef;
}

## blobs are raw bytes : decode valid UTF-8 for the text diff ##
for my $content_ref ( \$before_content, \$after_content ) {
    utf8::decode($$content_ref) if utf8::valid($$content_ref);
}

my @before_lines = split m{\n}, $before_content, -1;
my @after_lines  = split m{\n}, $after_content,  -1;

my ( $added_aref, $removed_aref )
    = <[base.diff_array]>->( \@before_lines, \@after_lines );

my @out = (
    "--- $before_cs",
    "+++ $after_cs",
    sprintf( '@@ -%d +%d @@', scalar @before_lines, scalar @after_lines )
);

push @out, map {"-$ARG"} $removed_aref->@*;
push @out, map {"+$ARG"} $added_aref->@*;

return join "\n", @out;
