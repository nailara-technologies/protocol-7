## task: flexible export stage selection + "since last export" filter

### context

`exportCSV()` and `showPrintTable()` in `index.html` have hardcoded stage
sets (`applied`, `interviewed`, `responded`, `rejected`). the user needs to
export any combination of stages — e.g. rejected-only, or all review items,
or only jobs not yet included in a previous export.

#### JS — stage checkbox generation

all known stages, in pipeline order:

```javascript
const ALL_EXPORT_STAGES = [
    'new', 'review', 'to_apply', 'applied', 'interviewed',
    'responded', 'rejected', 'skipped', 'archived', 'error'
];

const DEFAULT_EXPORT_STAGES = new Set(['applied','interviewed','responded','rejected']);
```

generate checkboxes from `ALL_EXPORT_STAGES`, pre-checked from
`DEFAULT_EXPORT_STAGES` (or from saved localStorage preference):

```javascript
function buildExportPanel() {
    const container = document.getElementById('export-stage-checks');
    container.innerHTML = '';
    const saved = loadExportPrefs();
    ALL_EXPORT_STAGES.forEach(s => {
        const checked = saved.stages.has(s) ? 'checked' : '';
        container.innerHTML += `
            <label style="display:flex;align-items:center;gap:0.25rem;cursor:pointer;">
              <input type="checkbox" class="export-stage-cb" value="${s}" ${checked}>
              <span style="color:#4a7090">${s}</span>
            </label>`;
    });
    document.getElementById('export-since-last').checked = saved.sinceLast;
    document.getElementById('export-current-filter').checked = saved.currentFilter;
}
```

#### JS — prefs persistence

```javascript
const EXPORT_PREFS_KEY = 'jobsite_export_prefs';
const LAST_EXPORT_KEY  = 'jobsite_last_export_ids';

function loadExportPrefs() {
    try {
        const p = JSON.parse(localStorage.getItem(EXPORT_PREFS_KEY) || '{}');
        return {
            stages:        new Set(p.stages || [...DEFAULT_EXPORT_STAGES]),
            sinceLast:     p.sinceLast     || false,
            currentFilter: p.currentFilter || false,
        };
    } catch(e) { return { stages: new Set(DEFAULT_EXPORT_STAGES),
                           sinceLast: false, currentFilter: false }; }
}

function saveExportPrefs() {
    const stages = [...document.querySelectorAll('.export-stage-cb:checked')]
        .map(cb => cb.value);
    const sinceLast     = document.getElementById('export-since-last').checked;
    const currentFilter = document.getElementById('export-current-filter').checked;
    localStorage.setItem(EXPORT_PREFS_KEY,
        JSON.stringify({ stages, sinceLast, currentFilter }));
    return { stages: new Set(stages), sinceLast, currentFilter };
}
```

### change 2: "since last export" filter

track which job IDs were included in the last export:

```javascript
function loadLastExportIds() {
    try { return new Set(JSON.parse(localStorage.getItem(LAST_EXPORT_KEY) || '[]')); }
    catch(e) { return new Set(); }
}

function saveLastExportIds(ids) {
    localStorage.setItem(LAST_EXPORT_KEY, JSON.stringify([...ids]));
}
```

when `sinceLast` is checked, add a filter: only include jobs whose ID is
NOT in `lastExportIds`. after export completes, save the exported IDs:

```javascript
function getExportRows(prefs) {
    const lastIds = prefs.sinceLast ? loadLastExportIds() : new Set();
    return sortedJobs()
        .filter(j => prefs.stages.has(j.stage))
        .filter(j => !prefs.sinceLast || !lastIds.has(j.id))
        .filter(j => !prefs.currentFilter || passesCurrentFilter(j));
}

function passesCurrentFilter(j) {
    ## applies the active tab + score filter same as render() ##
    if (filterMode !== 'alle' && j.stage !== filterMode) return false;
    const norm = parseFloat(j.score) > 10
        ? parseFloat(j.score)/10 : parseFloat(j.score);
    if (!isFinite(norm) || norm < minScore) return false;
    return true;
}
```

after a successful export, update last export IDs:
```javascript
const exported = rows.map(j => j.id);
saveLastExportIds(new Set([...loadLastExportIds(), ...exported]));
```

add a small indicator showing when the last export ran:
