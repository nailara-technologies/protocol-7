sub update_consciousness_model {
    my ($current_model, $new_observation) = @_;

    # Add to history
    push @{$current_model->{view_history}}, {
        %$new_observation,
        time => scalar(@{$current_model->{view_history}})
    };

    # Update velocities
    if (scalar(@{$current_model->{view_history}}) >= 2) {
        my $prev = $current_model->{view_history}->[-2];
        my $curr = $current_model->{view_history}->[-1];

        $current_model->{rotation_velocity} = $curr->{angle} - $prev->{angle};
        $current_model->{deepening_velocity} = $curr->{depth} - $prev->{depth};
    }

    # Update confidence trajectory
    $current_model->{learning_confidence} += 0.1 if $current_model->{learning_confidence} < 1.0;

    # Update handshake level
    my $views = scalar(@{$current_model->{view_history}});
    $current_model->{handshake_level} = min(6, 1 + int($views / 3));

    # Update immersion
    $current_model->{immersion_score} =
        ($current_model->{handshake_level} / 6.0) * $current_model->{learning_confidence};

    return $current_model;
}
