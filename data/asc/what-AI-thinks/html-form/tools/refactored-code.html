Refactored ncode Development Tool

#!/usr/bin/perl

# :: ncode :: development tool for parsing [perl\protocol-7] source code

use v5.30.0;
use strict;
use warnings;
use English;
use Term::ReadLine;
use Term::ANSIColor;
use File::Spec;
use IO::Select;
use Cwd qw(abs_path);
use Digest::BMW qw(bmw_256);
use List::MoreUtils qw(minmax uniq);
use Crypt::Misc qw(encode_b32r);
no warnings 'experimental';

# Configuration
BEGIN { $ENV{PERL_RL} = 'Perl' }

# Constants
use constant {
    SUCCESS => 0,
    ERROR => 1,
    DEFAULT_TARGET => '.',
    CONFIG_PATH => $ENV{HOME} . '/.code/',
};

# ANSI Color Configuration
our %COLORS = (
    T   => "\e[38;2;6;71;195m",    # Title
    B00 => "\e[48;2;0;0;17m",      # Background 0
    B01 => "\e[48;2;9;5;24m",      # Background 1
    B02 => "\e[48;2;9;5;42m",      # Background 2
    '0' => "\e[38;2;68;39;172m",   # Primary
    g   => "\e[38;2;71;195;6m",    # Success
    o   => "\e[38;2;197;141;7m",   # Warning
    B   => "\e[1m",                # Bold
    R   => "\e[0m",                # Reset
);

# State
my $USE_ANSI = 1;
my $term = Term::ReadLine->new('ncode');
$term->ornaments('0,0,md,me');

# Disable colors if requested
map { $COLORS{$_} = '' } keys %COLORS if !$USE_ANSI;

# File paths configuration
my $root_path = File::Spec->rel2abs(readlink($PROGRAM_NAME) // $PROGRAM_NAME);
$root_path =~ s|/[^/]+/[^/]+$||;    # relative to ./bin/ path [ parent ]

my $PATHS = {
    root => $root_path,
    work => CONFIG_PATH,
};

# Target paths for code parsing
my %TARGETS = (
    '.'      => Cwd::cwd(),
    bin      => "$PATHS->{root}/bin/",
    dev      => "$PATHS->{root}/bin/dev/",
    admin    => "$PATHS->{root}/bin/admin/",
    src      => "$PATHS->{root}/modules/",
    cfg      => "$PATHS->{root}/configuration/",
    md       => "$PATHS->{root}/read-me/md/",
    zenki    => "$PATHS->{root}/configuration/zenki/..,",
    'pm-src' => "$PATHS->{root}/data/lib-path/pm",
    amos     => "$PATHS->{root}/data/lib-path/pm/AMOS7",
    'amos-at'  => "$PATHS->{root}/data/lib-path/pm/AMOS7/Assert",
    'amos-in'  => "$PATHS->{root}/data/lib-path/pm/AMOS7/INLINE",
    'amos-cs'  => "$PATHS->{root}/data/lib-path/pm/AMOS7/CHKSUM",
    'amos-elf' => "$PATHS->{root}/data/lib-path/pm/AMOS7/CHKSUM/ELF",
);

# Command handlers
my %COMMANDS = (
    'list-targets'   => \&cmd_list_targets,
    'list-psubs'     => \&cmd_list_psubs,
    's'              => \&cmd_search,
    'search'         => \&cmd_search,
    'show-psub'      => \&cmd_show_psub,
    'move-psubs'     => \&cmd_move_psubs,
    'r'              => \&cmd_replace,
    'replace'        => \&cmd_replace,
    'parse'          => \&cmd_parse_headers,
    'parse-headers'  => \&cmd_parse_headers,
    'restore-backup' => \&cmd_restore_backup,
    'undo'           => \&cmd_undo,
    'undo-move'      => \&cmd_undo_move,
    'description'    => \&cmd_description
);

# Main
sub main {
    my ($cmd, @params) = @ARGV;

    $cmd //= '';

    # Default parameters
    push(@params, DEFAULT_TARGET) if $cmd =~ /^(?:s(?:earch)?|rep(?:lace)?)$/ && !@params;
    push(@params, 'src') if $cmd =~ /^parse(?:-headers)?$/ && !@params;

    # Setup signal handlers
    $SIG{INT} = sub {
        printf "%s%s\n\n", "\b" x 11, ' ' x 11;
        aborted();
    };

    print "\n";

    # Validate command
    if (!exists $COMMANDS{$cmd}) {
        if (length $cmd) {
            printf("\n %s::%s a command '%s%s%s' does not exist %s::%s\n",
                   $COLORS{o}, $COLORS{0}, $COLORS{o}, $cmd, $COLORS{0},
                   $COLORS{o}, $COLORS{R});
        } else {
            printf("\n %s::%s usage %s::%s\n",
                   $COLORS{0}, $COLORS{g}, $COLORS{0}, $COLORS{R});
        }
        cmd_description();
        exit ERROR;
    }

    # Handle 'all' target alias
    if (@params && $params[0] eq 'all') {
        $params[0] = join('+', grep { !/\./ } keys %TARGETS);
    }

    # Execute command
    if (ref $COMMANDS{$cmd} eq 'CODE') {
        $COMMANDS{$cmd}->(@params);
    } else {
        die "\n :: callback for command '$cmd' not defined\n\n";
    }

    exit SUCCESS;
}

# Utility Functions
sub highlight {
    my ($text, $pattern, $open_tag, $close_tag, $case_insensitive) = @_;
    $open_tag //= $COLORS{T};
    $close_tag //= $COLORS{R};
    $case_insensitive //= 1;

    if ($case_insensitive) {
        $text =~ s/($pattern)/$open_tag$1$close_tag/gi;
    } else {
        $text =~ s/($pattern)/$open_tag$1$close_tag/g;
    }
    return $text;
}

sub files {
    my ($type) = @_;
    my $path = $TARGETS{$type};
    my $recursion = $path =~ s/\.\.$// ? 1 : 0;

    die ":\n:: no such target :. $type\n\n" if !defined $path;
    die ":\n:: target path not exists :. $path\n\n" if !-e $path;

    my @files;
    if (-d $path) {
        opendir(my $dir_fh, $path) or die "cannot open dir '$path': $!\n";
        while (my $filename = readdir($dir_fh)) {
            next if $filename =~ /^\.{1,2}$/;
            $filename = "$path/$filename";
            $filename =~ s|//|/|g;

            if ($filename !~ /(?:\.bak|\.orig|\~)$/i) {
                push @files, $filename;

                if ($recursion && -d $filename) {
                    opendir(my $subdir_fh, $filename)
                        or die "cannot open dir '$filename': $!\n";
                    push @files, map { "$filename/$_" } readdir($subdir_fh);
                    closedir($subdir_fh);
                }
            }
        }
        closedir($dir_fh);
    }
    return sort @files;
}

# Entry point
main() unless caller;
